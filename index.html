```javascript // ПРОВЕРКА ДОСТИЖЕНИЙ ЭКОГРАДА function updateAchievements() { const count = Number( document.getElementById('plantCount')?.textContent || document.getElementById('profilePlantCount')?.textContent || 0 ); const achievements = [ { id: 'achievement5', goal: 5 }, { id: 'achievement15', goal: 15 }, { id: 'achievement25', goal: 25 } ]; achievements.forEach(achievement => { const element = document.getElementById(achievement.id); if (!element) return; if (count >= achievement.goal) { element.textContent = '🏆 Получено!'; element.classList.add('unlocked'); } else { element.textContent = 'Осталось: ' + (achievement.goal - count); element.classList.remove('unlocked'); } }); } window.updateAchievements = updateAchievements; document.addEventListener('DOMContentLoaded', updateAchievements); ```
